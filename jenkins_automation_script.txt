#!/bin/bash

# Java, Git, Maven
sudo dnf install -y java-21-amazon-corretto
java -version
sudo dnf install -y git maven

# Jenkins repo and install
sudo wget -O /etc/yum.repos.d/jenkins.repo https://pkg.jenkins.io/redhat-stable/jenkins.repo
sudo rpm --import https://pkg.jenkins.io/redhat-stable/jenkins.io-2023.key
sudo dnf install -y jenkins

# Use the disk for /tmp instead of RAM (takes effect after reboot)
sudo systemctl mask tmp.mount

# Give Jenkins its own temp directory on disk
sudo mkdir -p /var/lib/jenkins/tmp
sudo chown jenkins:jenkins /var/lib/jenkins/tmp

# Jenkins service override (replaces "systemctl edit jenkins")
sudo mkdir -p /etc/systemd/system/jenkins.service.d
sudo tee /etc/systemd/system/jenkins.service.d/override.conf > /dev/null <<'EOF'
[Service]
Environment="JAVA_OPTS=-Djava.awt.headless=true -Djava.io.tmpdir=/var/lib/jenkins/tmp"
EOF

# Reload, enable on boot, and start
sudo systemctl daemon-reload
sudo systemctl enable --now jenkins

# Reboot so /tmp moves to the disk; Jenkins starts again automatically
sudo reboot
